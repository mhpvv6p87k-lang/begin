    <!-- Application Script Logic -->
    <script>
        const SUPABASE_URL = 'https://vhgdiwrnvjfxplhcdulw.supabase.co';
        const SUPABASE_KEY = 'sb_publishable_0GpN4dmQn-b8CHxg_DovTg_rzzUoEaw';
        const supabaseClient = supabase.createClient(SUPABASE_URL, SUPABASE_KEY);

        // Your verified Gemini API Key from Supabase configuration
        const GEMINI_API_KEY = "482a25159b3c7e0f2685c6753a9708ff07d34ca2b2827a7cc161fa792629260b";

        let isSignUp = false;
        let currentUser = null;
        let userRole = 'student';

        window.addEventListener('load', () => {
            setTimeout(() => {
                document.getElementById('splash').classList.add('fade-out');
            }, 600);
            initApp();
        });

        async function initApp() {
            await checkUserSession();
            await loadFieldsDropdown();
            await loadOpportunities();
        }

        async function runDeepSearch() {
            const queryInput = document.getElementById('deepSearchInput');
            const resultBox = document.getElementById('deepSearchResult');
            const promptText = queryInput.value.trim();

            if (!promptText) return;

            resultBox.classList.remove('hidden');
            resultBox.innerHTML = '<div class="text-purple-300 animate-pulse flex items-center space-x-2"><span>Analyzing academic path and deep searching opportunities... ⏳</span></div>';

            try {
                const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=${GEMINI_API_KEY}`, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({
                        contents: [{ 
                            parts: [{ 
                                text: `You are an expert academic advisor and mentor for high school students in the Begin platform. Analyze the following request with deep analytical research and provide structured, clear point recommendations: ${promptText}` 
                            }] 
                        }]
                    })
                });

                const data = await response.json();
                const analysis = data.candidates?.[0]?.content?.parts?.[0]?.text || "Sorry, we could not generate an analysis at the moment. Please verify your API configuration.";

                resultBox.innerHTML = `
                    <div class="font-bold text-purple-300 mb-3 border-b border-purple-900/40 pb-2">Deep Search Analysis Results:</div>
                    <div>${analysis}</div>
                `;
            } catch (err) {
                resultBox.innerHTML = '<span class="text-red-400">An error occurred while connecting to the AI search engine. Please check your network or key.</span>';
            }
        }

        async function checkUserSession() {
            const { data: { session } } = await supabaseClient.auth.getSession();
            currentUser = session?.user || null;
            if (currentUser) {
                const { data } = await supabaseClient.from('user_roles').select('role').eq('user_id', currentUser.id).single();
                userRole = data?.role || 'student';
            }
            renderNav();
            toggleAdminView();
        }

        function renderNav() {
            const nav = document.getElementById('authNav');
            if (currentUser) {
                nav.innerHTML = `
                    <span class="text-xs text-purple-300 hidden sm:inline">Hi, ${currentUser.email.split('@')[0]}</span>
                    <button onclick="handleSignOut()" class="bg-purple-950 hover:bg-purple-900 border border-purple-800/50 px-4 py-2 rounded-xl text-sm font-medium transition">Sign Out</button>
                `;
            } else {
                nav.innerHTML = `
                    <button onclick="toggleAuthModal(true)" class="bg-gradient-to-r from-purple-600 to-indigo-600 hover:from-purple-500 hover:to-indigo-500 px-4 py-2 rounded-xl text-sm font-semibold transition">Sign In</button>
                `;
            }
        }

        function toggleAdminView() {
            const panel = document.getElementById('adminPanel');
            if (userRole === 'admin') {
                panel.classList.remove('hidden');
            } else {
                panel.classList.add('hidden');
            }
        }

        function toggleAuthModal(show) {
            const modal = document.getElementById('authModal');
            if (show) modal.classList.remove('hidden');
            else modal.classList.add('hidden');
        }

        function switchAuthMode() {
            isSignUp = !isSignUp;
            document.getElementById('modalTitle').innerText = isSignUp ? 'Create Account' : 'Sign In to Begin';
            document.getElementById('authSwitchText').innerText = isSignUp ? 'Already have an account?' : "Don't have an account?";
        }

        document.getElementById('authForm').addEventListener('submit', async (e) => {
            e.preventDefault();
            const email = document.getElementById('authEmail').value;
            const password = document.getElementById('authPassword').value;

            if (isSignUp) {
                const { error } = await supabaseClient.auth.signUp({ email, password });
                if (error) alert(error.message);
                else { alert('Account created successfully! You can now sign in.'); switchAuthMode(); }
            } else {
                const { error } = await supabaseClient.auth.signInWithPassword({ email, password });
                if (error) alert(error.message);
                else { toggleAuthModal(false); window.location.reload(); }
            }
        });

        async function handleSignOut() {
            await supabaseClient.auth.signOut();
            window.location.reload();
        }

        async function loadFieldsDropdown() {
            const { data } = await supabaseClient.from('fields').select('*');
            const select = document.getElementById('oppField');
            if (data) {
                select.innerHTML = '<option value="">Select Field</option>';
                data.forEach(f => {
                    const opt = document.createElement('option');
                    opt.value = f.id;
                    opt.textContent = f.name;
                    select.appendChild(opt);
                });
            }
        }

        async function loadOpportunities() {
            const { data } = await supabaseClient.from('opportunities').select('*, fields(name)');
            const grid = document.getElementById('opportunitiesGrid');
            grid.innerHTML = '';
            if (data && data.length > 0) {
                data.forEach(opp => {
                    grid.innerHTML += `
                        <div class="bg-slate-900/60 border border-purple-900/30 rounded-2xl p-6 flex flex-col justify-between hover:border-purple-500/50 transition shadow-lg">
                            <div>
                                <span class="text-xs px-3 py-1 rounded-full bg-purple-950 text-purple-300 border border-purple-800/40 font-medium">${opp.fields?.name || 'General'}</span>
                                <h4 class="text-xl font-bold text-white mt-3 mb-2">${escapeHtml(opp.title)}</h4>
                                <p class="text-slate-400 text-sm mb-4 leading-relaxed">${escapeHtml(opp.description)}</p>
                            </div>
                            <a href="${escapeHtml(opp.apply_url)}" target="_blank" class="inline-flex items-center justify-center w-full bg-purple-950/60 hover:bg-purple-900 text-purple-200 text-sm font-semibold py-2.5 px-4 rounded-xl border border-purple-800/50 transition">Apply Now &rarr;</a>
                        </div>
                    `;
                });
            } else {
                grid.innerHTML = `<p class="col-span-full text-center text-slate-500 py-10">No opportunities posted yet. Use the admin panel above to add some!</p>`;
            }
        }

        document.getElementById('addOppForm').addEventListener('submit', async (e) => {
            e.preventDefault();
            const title = document.getElementById('oppTitle').value;
            const field_id = document.getElementById('oppField').value;
            const apply_url = document.getElementById('oppUrl').value;
            const description = document.getElementById('oppDesc').value;

            const { error } = await supabaseClient.from('opportunities').insert([{ title, field_id, apply_url, description }]);
            if (error) alert(error.message);
            else {
                alert('Opportunity published successfully!');
                document.getElementById('addOppForm').reset();
                loadOpportunities();
            }
        });

        document.getElementById('submissionForm').addEventListener('submit', async (e) => {
            e.preventDefault();
            const submitter_name = document.getElementById('subName').value;
            const title = document.getElementById('subOppTitle').value;
            const apply_url = document.getElementById('subUrl').value;
            const social_contact = document.getElementById('subSocial').value;

            const description = `Submitted by: ${submitter_name} | Contact: ${social_contact}`;
            const { error } = await supabaseClient.from('opportunities').insert([{ title, apply_url, description }]);
            
            if (error) {
                alert('Error submitting: ' + error.message);
            } else {
                alert('Thank you! Your opportunity has been submitted for review.');
                document.getElementById('submissionForm').reset();
                loadOpportunities();
            }
        });

        function escapeHtml(str) {
            return str.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;").replace(/'/g, "&#039;");
        }
